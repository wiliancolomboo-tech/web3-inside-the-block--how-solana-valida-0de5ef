# Multi-stage enterprise build for the Solana Block Ordering Daemon
FROM rust:1.77-bullseye AS rust-builder
WORKDIR /usr/src/solana-block-orderer

RUN apt-get update && apt-get install -y cmake protobuf-compiler libssl-dev pkg-config && rm -rf /var/lib/apt/lists/*

COPY Cargo.toml Cargo.lock ./
COPY src ./src
COPY tests ./tests

# Validate test assertions in container build
RUN cargo test --release -- --nocapture
RUN cargo build --release

# Stage 2: Node.js Telemetry Worker
FROM node:20-bullseye-slim AS node-runner
WORKDIR /app

RUN apt-get update && apt-get install -y ca-certificates curl && rm -rf /var/lib/apt/lists/*

COPY package.json tsconfig.json ./
RUN npm install
COPY src ./src
RUN npm run build

# Copy Rust native binary
COPY --from=rust-builder /usr/src/solana-block-orderer/target/release/solana-block-orderer /usr/local/bin/

ENV NODE_ENV=production
EXPOSE 8080 9100

HEALTHCHECK --interval=15s --timeout=5s --start-period=5s --retries=3 \
  CMD curl -f http://localhost:8080/health || exit 1

CMD ["npm", "start"]
