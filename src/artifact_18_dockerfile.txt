# Multi-stage build for Rust Ordering Engine & TS Daemon
FROM rust:1.77-slim-bullseye AS rust-builder
WORKDIR /app/rust
COPY solana-ordering-engine/Cargo.toml ./
COPY solana-ordering-engine/src ./src
RUN cargo build --release

FROM node:20-bullseye-slim AS node-builder
WORKDIR /app/ts
COPY orderflow-monitor/package*.json ./
COPY orderflow-monitor/tsconfig.json ./
RUN npm ci
COPY orderflow-monitor/src ./src
RUN npm run build

FROM debian:bullseye-slim AS runner
WORKDIR /opt/solana-orderflow
RUN apt-get update && apt-get install -y --no-install-recommends ca-certificates curl nodejs npm && rm -rf /var/lib/apt/lists/*

COPY --from=rust-builder /app/rust/target/release/solana-ordering-engine /usr/local/bin/solana-ordering-engine
COPY --from=node-builder /app/ts/dist ./dist
COPY --from=node-builder /app/ts/node_modules ./node_modules
COPY --from=node-builder /app/ts/package.json ./package.json

EXPOSE 9090
HEALTHCHECK --interval=15s --timeout=5s --start-period=5s --retries=3 \
  CMD curl -f http://localhost:9090/health || exit 1

CMD ["node", "dist/index.js"]
