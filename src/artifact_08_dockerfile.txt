FROM rust:1.78-bullseye AS rust-builder
WORKDIR /usr/src/sbe-core
COPY sbe-core/Cargo.toml sbe-core/Cargo.lock* ./
RUN mkdir src && echo "fn main() {}" > src/main.rs && cargo build --release || true
COPY sbe-core/src ./src
RUN cargo test --release

FROM node:20-bullseye-slim AS node-runner
WORKDIR /app
COPY sbe-agent/package*.json ./
RUN npm install
COPY sbe-agent/tsconfig.json ./
COPY sbe-agent/src ./src
RUN npm run build

ENV NODE_ENV=production
HEALTHCHECK --interval=30s --timeout=5s --start-period=5s --retries=3 \
  CMD node -e "require('http').get('http://localhost:8080/health', (r) => {if (r.statusCode !== 200) process.exit(1);})"

EXPOSE 8080
ENTRYPOINT ["node", "dist/cli.js"]
CMD ["monitor", "--rpc", "https://api.mainnet-beta.solana.com"]
