FROM rust:1.75-bullseye AS builder
WORKDIR /usr/src/solana-order-engine
COPY Cargo.toml ./
COPY src ./src
RUN cargo build --release

FROM debian:bullseye-slim
RUN apt-get update && apt-get install -y ca-certificates && rm -rf /var/lib/apt/lists/*
COPY --from=builder /usr/src/solana-order-engine/target/release/solana-order-engine /usr/local/bin/
CMD ["solana-order-engine"]
